   283	
   284	      setToasts((prev) => {
   285	        // Prevent duplicate spam of identical message
   286	        const filtered = prev.filter((t) => t.message !== msg);
   287	        // Prepend new toast and keep maximum 3 in the stack
   288	        return [newToast, ...filtered].slice(0, 3);
   289	      });
   290	
   291	      // Auto dismiss after duration
   292	      const timer = setTimeout(() => {
   293	        setToasts((current) => current.filter((t) => t.id !== id));
   294	      }, duration);
   295	
   296	      return () => clearTimeout(timer);
   297	    });
   298	
   299	    return unsub;
   300	  }, []);
   301	
   302	  if (toasts.length === 0) return null;
   303	
   304	  return (
   305	    <div
   306	      role="region"
   307	      aria-label="System Notifications"
   308	      className="fixed bottom-6 left-0 right-0 mx-auto w-full max-w-sm px-4 pointer-events-none z-[9999] h-20 relative flex items-end justify-center select-none"
   309	    >
   310	      {toasts.map((item, index) => (
   311	        <StackedToastCard
   312	          key={item.id}
   313	          item={item}
   314	          index={index}
   315	          total={toasts.length}
   316	          onDismiss={handleDismiss}
   317	        />
   318	      ))}
   319	    </div>
   320	  );
   321	}

   596	export interface OfflineFlushResult {
   597	  syncedCount: number;
   598	  failedCount: number;
   599	  authRequired?: boolean;
   600	}
   601	
   602	let activeMutationsFlush: Promise<OfflineFlushResult> | null = null;
   603	let activeQueueFlush: Promise<OfflineFlushResult> | null = null;
   604	let pendingQueueFlushRequested = false;
   605	
   606	// Flush all queued non-score mutations (events, participants, judges, orgs) to Supabase
   607	export function flushOfflineMutationsToSupabase(): Promise<OfflineFlushResult> {
   608	  if (activeMutationsFlush) {
   609	    return activeMutationsFlush;
   610	  }
   611	  activeMutationsFlush = runFlushOfflineMutationsOnce().finally(() => {
   612	    activeMutationsFlush = null;
   613	  });
   614	  return activeMutationsFlush;
   615	}
   616	
   617	async function runFlushOfflineMutationsOnce(): Promise<OfflineFlushResult> {
   618	  if (typeof navigator !== 'undefined' && !navigator.onLine) {
   619	    return { syncedCount: 0, failedCount: 0 };
   620	  }
   621	  if (!isSupabaseConfigured()) {
   622	    return { syncedCount: 0, failedCount: 0 };
   623	  }
   624	
   625	  const rawMutations = (await getPendingMutations()).filter((m) => m.syncStatus !== 'failed');
   626	  if (rawMutations.length === 0) {
   627	    return { syncedCount: 0, failedCount: 0 };
   628	  }
   629	
   630	  // 1. Client-Side Optimistic Concurrency Control (OCC) Deduplication
   631	  const latestMutationMap = new Map<string, PendingMutationRecord>();
   632	  const obsoleteMutationIds: string[] = [];
   633	
   634	  rawMutations.sort((a, b) => new Date(a.queuedAt).getTime() - new Date(b.queuedAt).getTime());
   635	
   636	  const db = await getOfflineDB();
   637	
   638	  for (const item of rawMutations) {
   639	    const key = `${item.table}:${item.entityId}`;
   640	    if (latestMutationMap.has(key)) {
   641	      const prev = latestMutationMap.get(key)!;
   642	      if (prev.operation === 'INSERT' && item.operation === 'UPDATE') {
   643	        prev.payload = { ...(prev.payload || {}), ...(item.payload || {}) };
   644	        prev.queuedAt = item.queuedAt;
   645	        await db.put('outbox', mutationToOutbox(prev));
   646	        obsoleteMutationIds.push(item.id);
   647	        continue;
   648	      }
   649	      if (prev.operation === 'UPDATE' && item.operation === 'UPDATE') {
   650	        item.payload = { ...(prev.payload || {}), ...(item.payload || {}) };
   651	        await db.put('outbox', mutationToOutbox(item));
   652	        obsoleteMutationIds.push(prev.id);
   653	        latestMutationMap.set(key, item);
   654	        continue;
   655	      }
   656	      obsoleteMutationIds.push(prev.id);
   657	    }
   658	    latestMutationMap.set(key, item);
   659	  }
   660	
   661	  if (obsoleteMutationIds.length > 0) {
   662	    await Promise.all(obsoleteMutationIds.map((id) => removePendingMutation(id)));
   663	  }
   664	
   665	  const mutations = Array.from(latestMutationMap.values()).filter((m) => m.syncStatus !== 'failed');
   666	  let syncedCount = 0;
   667	  let failedCount = 0;
   668	  let authRequired = false;
   669	
   670	  for (const item of mutations) {
   671	    try {
   672	      if (item.table === 'events') {
   673	        if (item.operation === 'INSERT') {
   674	          await callSecuredWriteRpc('event_upsert', { p_record: item.payload });
   675	        } else if (item.operation === 'UPDATE') {
   676	          await callSecuredWriteRpc('event_update', {
   677	            p_event_id: item.entityId,
   678	            p_updates: item.payload,
   679	          });
   680	        } else if (item.operation === 'DELETE') {
   681	          await callSecuredWriteRpc('event_delete', { p_event_id: item.entityId });
   682	        }
   683	      } else if (item.table === 'participants') {
   684	        if (item.operation === 'INSERT') {
   685	          const records = Array.isArray(item.payload) ? item.payload : [item.payload];
   686	          await callSecuredWriteRpc('participant_upsert_batch', { p_records: records });
   687	        } else if (item.operation === 'UPDATE') {
   688	          await callSecuredWriteRpc('participant_update', {
   689	            p_participant_id: item.entityId,
   690	            p_updates: item.payload,
   691	          });
   692	        } else if (item.operation === 'DELETE') {
   693	          await callSecuredWriteRpc('participant_delete', { p_participant_id: item.entityId });
   694	        }
   695	      } else if (item.table === 'judges') {
   696	        if (item.operation === 'INSERT') {
   697	          await callSecuredWriteRpc('judge_upsert', { p_record: item.payload });
   698	        } else if (item.operation === 'UPDATE') {
   699	          await callSecuredWriteRpc('judge_update', {
   700	            p_judge_id: item.entityId,
   701	            p_updates: item.payload,
   702	          });
   703	        } else if (item.operation === 'DELETE') {
   704	          await callSecuredWriteRpc('judge_delete', { p_judge_id: item.entityId });
   705	        }
   706	      } else if (item.table === 'organizations') {
   707	        if (item.operation === 'UPDATE') {
   708	          await callSecuredWriteRpc('org_update', {
   709	            p_org_id: item.entityId,
   710	            p_updates: item.payload,
   711	          });
   712	        } else if (item.operation === 'DELETE') {
   713	          await callSecuredWriteRpc('org_delete', { p_org_id: item.entityId });
   714	        }
   715	      }
   716	
   717	      await removePendingMutation(item.id);
   718	      syncedCount++;
   719	    } catch (err: any) {
   720	      console.warn(`Error syncing offline mutation ${item.id} (${item.table}):`, err?.message || err);
   721	
   722	      if (err?.code === 'unauthorized') {
   723	        authRequired = true;
   724	        break;
   725	      }
   726	
   727	      if (err?.code === 'rpc_unavailable') {
   728	        break;
   729	      }
   730	
   731	      const orgBlock = parseOrgEnforcementError(err);
   732	      if (orgBlock) {
   733	        eventBus.emit('org_enforcement', { code: orgBlock });
   734	        item.lastError = `ORG_NOT_ALLOWED: ${orgBlock}`;
   735	        if (orgBlock === 'SUSPENDED' || orgBlock === 'DELETED') {
   736	          item.syncStatus = 'failed';
   737	          blackBox.recordError(
   738	            'console_error',
   739	            `Offline ${item.table}:${item.operation} blocked (${orgBlock})`,
   740	            'offlineQueue.mutations'
   741	          );
   742	        }
   743	        await db.put('outbox', mutationToOutbox(item));
   744	        failedCount++;
   745	        continue;
   746	      }
   747	
   748	      const isRosterLocked = Boolean(
   749	        err?.code === 'ROSTER_LOCKED' ||
   750	        err?.message?.includes('ROSTER_LOCKED') ||
   751	        (err?.details && String(err.details).includes('ROSTER_LOCKED'))
   752	      );
   753	      const isRls = Boolean(
   754	        err?.code === '42501' ||
   755	        err?.code === 'forbidden' ||
   756	        err?.message?.toLowerCase().includes('row-level security') ||
   757	        err?.message?.toLowerCase().includes('permission denied')
   758	      );
   759	      const isServerReturned = Boolean(err?.code || isRosterLocked || isRls);
   760	
   761	      if (isRosterLocked && item.table === 'events' && item.operation === 'UPDATE' && item.payload) {
   762	        const fullPayload = { ...(item.payload || {}) };
   763	        const rosterPayload: Record<string, any> = {};
   764	        const nonRosterPayload: Record<string, any> = {};
   765	
   766	        for (const [k, v] of Object.entries(fullPayload)) {
   767	          if (k === 'assigned_participants' || k === 'assigned_judges' || k === 'participant_sl_numbers') {
   768	            rosterPayload[k] = v;
   769	          } else {
   770	            nonRosterPayload[k] = v;
   771	          }
   772	        }
   773	
   774	        if (Object.keys(nonRosterPayload).length > 0) {
   775	          const newNonRosterItem: PendingMutationRecord = {
   776	            id: `mut_${Date.now()}_${Math.random().toString(36).substring(2, 9)}`,
   777	            table: item.table,
   778	            operation: 'UPDATE',
   779	            entityId: item.entityId,
   780	            payload: nonRosterPayload,
   781	            queuedAt: new Date().toISOString(),
   782	            syncStatus: 'pending',
   783	            retryCount: 0,
   784	          };
   785	          await db.put('outbox', mutationToOutbox(newNonRosterItem));
   786	        }
   787	
   788	        item.payload = rosterPayload;
   789	        item.syncStatus = 'failed';
   790	        item.lastError = 'ROSTER_LOCKED: Roster modifications rejected because event is locked.';
   791	        await db.put('outbox', mutationToOutbox(item));
   792	        failedCount++;
   793	        continue;
   794	      }
   795	
   796	      if (!isServerReturned) {
   797	        failedCount++;
   798	        continue;
   799	      }
   800	
   801	      item.retryCount = (item.retryCount || 0) + 1;
   802	      item.lastError = err?.message || 'Server rejected mutation';
   803	
   804	      const isRosterLockedDelete =
   805	        isRosterLocked && (item.table === 'participants' || item.table === 'judges') && item.operation === 'DELETE';
   806	
   807	      if (item.retryCount >= 3 || isRls || isRosterLockedDelete) {
   808	        item.syncStatus = 'failed';
   809	        blackBox.recordError(
   810	          'console_error',
   811	          `Offline ${item.table}:${item.operation} failed: ${item.lastError}`,
   812	          'offlineQueue.mutations'
   813	        );
   814	      }
   815	
   816	      await db.put('outbox', mutationToOutbox(item));
   817	      failedCount++;
   818	    }
   819	  }
   820	
   821	  notifyQueueChange();
   822	  return { syncedCount, failedCount, authRequired };
   823	}

   829	export async function queueScoreLocally(score: any): Promise<string> {
   830	  const db = await getOfflineDB();
   831	  const id =
   832	    score.id ||
   833	    (typeof crypto !== 'undefined' && crypto.randomUUID
   834	      ? crypto.randomUUID()
   835	      : `score_${Date.now()}_${Math.random().toString(36).substring(2, 9)}`);
   836	
   837	  // Idempotency (Double-Submission Prevention)
   838	  const existing = await db.get('outbox', id);
   839	  if (existing && existing.kind === 'score') {
   840	    return id;
   841	  }
   842	
   843	  const eventId = score.event_id || score.eventId || '';
   844	  const judgeId = score.judge_id || score.judgeId || '';
   845	  const participantId = score.participant_id || score.participantId || '';
   846	  const scoreData = score.score_data || score.scoreData || score.score_payload?.criteria || {};
   847	  const totalScore = Number(score.total_score !== undefined ? score.total_score : score.totalScore) || 0;
   848	
   849	  const scoreRec: PendingScoreRecord = {
   850	    id,
   851	    event_id: eventId,
   852	    judge_id: judgeId,
   853	    participant_id: participantId,
   854	    score_data: scoreData,
   855	    total_score: totalScore,
   856	    score_payload: score.score_payload || {
   857	      criteria: scoreData,
   858	      total_score: totalScore,
   859	      submitted_at: new Date().toISOString(),
   860	    },
   861	    queuedAt: new Date().toISOString(),
   862	    syncStatus: 'pending',
   863	    retryCount: 0,
   864	  };
   865	
   866	  await db.put('outbox', scoreToOutbox(scoreRec));
   867	  notifyQueueChange();
   868	  requestBackgroundSync();
   869	
   870	  getPendingCount()
   871	    .then((count) => {
   872	      eventBus.emit('sync_status', {
   873	        status: 'offline',
   874	        pendingCount: count,
   875	        message: `${count} item${count > 1 ? 's' : ''} queued locally`,
   876	      });
   877	    })
   878	    .catch(() => {});
   879	
   880	  return id;
   881	}

   982	async function handleScoreSyncError(
   983	  db: IDBPDatabase<FairJudgeOfflineDB>,
   984	  item: PendingScoreRecord,
   985	  err: any
   986	): Promise<{ authRequired?: boolean; rpcUnavailable?: boolean; newlyFailed?: boolean }> {
   987	  if (err?.code === 'unauthorized') {
   988	    return { authRequired: true };
   989	  }
   990	  if (err?.code === 'rpc_unavailable') {
   991	    return { rpcUnavailable: true };
   992	  }
   993	  const orgBlock = parseOrgEnforcementError(err);
   994	  if (orgBlock) {
   995	    eventBus.emit('org_enforcement', { code: orgBlock });
   996	    item.lastError = `ORG_NOT_ALLOWED: ${orgBlock}`;
   997	    let newlyFailed = false;
   998	    if (orgBlock === 'SUSPENDED' || orgBlock === 'DELETED') {
   999	      item.syncStatus = 'failed';
  1000	      newlyFailed = true;
  1001	      blackBox.recordError(
  1002	        'console_error',
  1003	        `Offline ballot sync blocked (${orgBlock})`,
  1004	        'offlineQueue.scores'
  1005	      );
  1006	    }
  1007	    await db.put('outbox', scoreToOutbox(item)).catch(() => {});
  1008	    return { newlyFailed };
  1009	  }
  1010	
  1011	  const errMsg = String(err?.message || err || 'Server rejected ballot');
  1012	  const lowerMsg = errMsg.toLowerCase();
  1013	  const isPermanentReject = Boolean(
  1014	    err?.code === '42501' ||
  1015	      err?.code === 'forbidden' ||
  1016	      err?.code === 'not_found' ||
  1017	      err?.code === 'ROSTER_LOCKED' ||
  1018	      err?.code === 'RLS Permission Denied' ||
  1019	      err?.code === '23505' ||
  1020	      err?.code === '23503' ||
  1021	      err?.code === '23502' ||
  1022	      err?.code === '22P02' ||
  1023	      errMsg.includes('RLS Permission Denied') ||
  1024	      lowerMsg.includes('permission denied') ||
  1025	      lowerMsg.includes('row-level security') ||
  1026	      lowerMsg.includes('not assigned') ||
  1027	      lowerMsg.includes('event has ended') ||
  1028	      lowerMsg.includes('event is paused') ||
  1029	      lowerMsg.includes('ballot submission rejected')
  1030	  );
  1031	  const isServerReturned = Boolean(err?.code || isPermanentReject);
  1032	
  1033	  if (!isServerReturned) {
  1034	    return {};
  1035	  }
  1036	
  1037	  item.retryCount = (item.retryCount || 0) + 1;
  1038	  item.lastError = errMsg;
  1039	  let newlyFailed = false;
  1040	  if (isPermanentReject || item.retryCount >= 3) {
  1041	    item.syncStatus = 'failed';
  1042	    newlyFailed = true;
  1043	    blackBox.recordError(
  1044	      'console_error',
  1045	      `Offline ballot rejected: ${errMsg}`,
  1046	      'offlineQueue.scores'
  1047	    );
  1048	  }
  1049	  await db.put('outbox', scoreToOutbox(item)).catch(() => {});
  1050	  return { newlyFailed };
  1051	}

  1053	// Sync all pending scores and mutations from unified outbox to Supabase
  1054	export function flushOfflineQueueToSupabase(): Promise<OfflineFlushResult> {
  1055	  if (activeQueueFlush) {
  1056	    pendingQueueFlushRequested = true;
  1057	    return activeQueueFlush;
  1058	  }
  1059	
  1060	  activeQueueFlush = (async () => {
  1061	    let aggregateSynced = 0;
  1062	    let lastResult: OfflineFlushResult = { syncedCount: 0, failedCount: 0 };
  1063	
  1064	    do {
  1065	      pendingQueueFlushRequested = false;
  1066	      lastResult = await runFlushOfflineQueueOnce();
  1067	      aggregateSynced += lastResult.syncedCount;
  1068	      if (lastResult.authRequired) {
  1069	        pendingQueueFlushRequested = false;
  1070	        break;
  1071	      }
  1072	    } while (pendingQueueFlushRequested);
  1073	
  1074	    return {
  1075	      syncedCount: aggregateSynced,
  1076	      failedCount: lastResult.failedCount,
  1077	      authRequired: lastResult.authRequired,
  1078	    };
  1079	  })().finally(() => {
  1080	    activeQueueFlush = null;
  1081	  });
  1082	
  1083	  return activeQueueFlush;
  1084	}
  1085	
  1086	async function runFlushOfflineQueueOnce(): Promise<OfflineFlushResult> {
  1087	  if (typeof navigator !== 'undefined' && !navigator.onLine) {
  1088	    return { syncedCount: 0, failedCount: 0 };
  1089	  }
  1090	
  1091	  // 1. Flush offline structural mutations first (events, participants, judges)
  1092	  const mutationResult = await flushOfflineMutationsToSupabase();
  1093	  let authRequired = Boolean(mutationResult.authRequired);
  1094	
  1095	  // 2. Flush pending scores (ballots) with OCC deduplication and batched upsert
  1096	  const rawScores = (await getPendingScores()).filter((s) => s.syncStatus !== 'failed');
  1097	  let scoreSynced = 0;
  1098	  let scoreFailed = 0;
  1099	
  1100	  if (!authRequired && isSupabaseConfigured() && rawScores.length > 0) {
  1101	    const dedupedScoreMap = new Map<string, PendingScoreRecord>();
  1102	    const obsoleteScoreIds: string[] = [];
  1103	
  1104	    rawScores.sort((a, b) => new Date(a.queuedAt).getTime() - new Date(b.queuedAt).getTime());
  1105	
  1106	    for (const item of rawScores) {

    74	  key: string,
    75	  force: boolean,
    76	  loader: () => Promise<ReadResult<T>>

   358	let lastClockSyncAt = 0;
   359	let inflightClockSync: Promise<number> | null = null;
   360	
   361	export async function syncServerClockOffset(force: boolean = false): Promise<number> {
   362	  if (!isSupabaseConfigured() || !isOnlineNavigator()) {
   363	    return getServerClockOffsetMs();
   364	  }
   365	  const nowMono = Date.now();
   366	  if (!force && lastClockSyncAt > 0 && nowMono - lastClockSyncAt < 60_000) {
   367	    return getServerClockOffsetMs();
   368	  }
   369	  if (inflightClockSync) return inflightClockSync;
   370	
   371	  inflightClockSync = (async () => {
   372	    try {
   373	      const tBefore = Date.now();
   374	      const { data, error } = await withReadTimeout<{ data: any; error: any }>(
   375	        (supabase.rpc as any)('fj_server_time'),
   376	        'Sync server clock',
   377	        5000
   378	      );
   379	      const tAfter = Date.now();
   380	      if (!error && data) {
   381	        const serverMs = Date.parse(String(data));
   382	        if (Number.isFinite(serverMs)) {
   383	          const clientMidMs = (tBefore + tAfter) / 2;
   384	          const offsetMs = serverMs - clientMidMs;
   385	          setServerClockOffsetMs(offsetMs);
   386	          lastClockSyncAt = Date.now();
   387	          return offsetMs;
   388	        }
   389	      }
   390	    } catch (err) {
   391	      console.warn('Server clock sync note:', err);
   392	    } finally {
   393	      inflightClockSync = null;
   394	    }
   395	    return getServerClockOffsetMs();
   396	  })();
   397	
   398	  return inflightClockSync;
   399	}

   401	/**
   402	 * Executes a Stage D3 / 15.10 token-authenticated write RPC exclusively.
   403	 * Never falls back to direct table writes.
   404	 * Emits `session_expired` when online with a missing/expired token.
   405	 */
   406	export async function callSecuredWriteRpc<T = any>(
   407	  rpcName: string,
   408	  params: Record<string, any>
   409	): Promise<{ data?: T }> {
   410	  const token = getStoredSessionToken() || '';
   411	  if (!token) {
   412	    eventBus.emit('session_expired', {
   413	      role: isAuthRequiredSessionRole(),
   414	      reason: 'missing',
   415	    });
   416	    const authErr = new Error('Session token missing. Please verify your credentials to continue.');
   417	    (authErr as any).code = 'unauthorized';
   418	    throw authErr;
   419	  }
   420	
   421	  const { data, error } = await withReadTimeout(
   422	    supabase.rpc(rpcName, {
   423	      p_token: token,
   424	      ...params,
   425	    }),
   426	    `Write RPC (${rpcName})`,
   427	    20_000
   428	  );
   429	
   430	  if (error) {
   431	    const msg = (error.message || '').toLowerCase();
   432	    if (error.code === 'PGRST202' || error.code === '42883' || msg.includes(`function public.${rpcName}`)) {
   433	      const rpcErr = new Error('Database security RPCs are not deployed. Please run sql/15.10-stage1-security-lock.sql.');
   434	      (rpcErr as any).code = 'rpc_unavailable';
   435	      throw rpcErr;
   436	    }
   437	    throw error;
   438	  }
   439	
   440	  const res = data as any;
   441	  if (!res || !res.ok) {
   442	    const errCode = res?.error || 'RPC_ERROR';
   443	    const msg = res?.message || res?.error || 'Database operation rejected.';
   444	    if (errCode === 'unauthorized') {
   445	      eventBus.emit('session_expired', {
   446	        role: isAuthRequiredSessionRole(),
   447	        reason: 'expired',
   448	      });
   449	    }
   450	    const err = new Error(msg);
   451	    (err as any).code = errCode === 'ROSTER_LOCKED' ? 'ROSTER_LOCKED' : errCode;
   452	    throw err;
   453	  }
   454	
   455	  return { data: res.data as T };
   456	}

   458	/**
   459	 * Executes a Phase 2A token-authenticated read RPC exclusively.
   460	 * Rejects unauthenticated callers when no session token is available.
   461	 */
   462	export async function callSecuredReadRpc<T = any>(
   463	  rpcName: string,
   464	  params: Record<string, any> = {},
   465	  tokenOverride?: string
   466	): Promise<{ data?: T; not_found?: boolean; can_delete?: boolean; judges?: any[]; participants?: any[] }> {
   467	  const token = (tokenOverride || getStoredSessionToken() || '').trim();
   468	  if (!token) {
   469	    const authErr = new Error('Session expired or missing. Please sign in again.');
   470	    (authErr as any).code = 'unauthorized';
   471	    throw authErr;
   472	  }
   473	
   474	  const { data, error } = await withReadTimeout(
   475	    supabase.rpc(rpcName, {
   476	      p_token: token,
   477	      ...params,
   478	    }),
   479	    `Read RPC (${rpcName})`,
   480	    READ_TIMEOUT_MS
   481	  );
   482	
   483	  if (error) {
   484	    const msg = (error.message || '').toLowerCase();
   485	    if (error.code === 'PGRST202' || error.code === '42883' || msg.includes(`function public.${rpcName}`)) {
   486	      const rpcErr = new Error(
   487	        'Database read RPCs are not deployed yet. Please run sql/15.12-phase2a-read-rpcs.sql.'
   488	      );
   489	      (rpcErr as any).code = 'rpc_unavailable';
   490	      throw rpcErr;
   491	    }
   492	    throw error;
   493	  }
   494	
   495	  const res = data as any;
   496	  if (!res || !res.ok) {
   497	    const errCode = res?.error || 'RPC_ERROR';
   498	    const msg = res?.message || res?.error || 'Database read rejected.';
   499	    const err = new Error(msg);
   500	    (err as any).code = errCode;
   501	    throw err;
   502	  }
   503	
   504	  return res;
   505	}

   549	const DB_CHANGE_EVENT = 'fairjudge_db_change';
   550	let debounceTimer: ReturnType<typeof setTimeout> | null = null;
   551	let batchedTables = new Set<DbTableName>();
   552	
   553	// Debounced event broadcaster: batches multiple database changes into EventBus & CustomEvents
   554	export function notifyDatabaseChange(table: string) {
   555	  batchedTables.add(normalizeDbTableName(table));
   556	  if (debounceTimer) {
   557	    clearTimeout(debounceTimer);
   558	  }
   559	  debounceTimer = setTimeout(() => {
   560	    const tables = Array.from(batchedTables);
   561	    batchedTables.clear();
   562	    const emittedTable: DbTableName = tables.length === 1 ? tables[0] : DB_TABLES.ALL;
   563	
   564	    // Broadcast through typed EventBus
   565	    eventBus.emit('cache_updated', { table: emittedTable });
   566	
   567	    if (typeof window !== 'undefined') {
   568	      window.dispatchEvent(
   569	        new CustomEvent(DB_CHANGE_EVENT, {
   570	          detail: { table: emittedTable, tables },
   571	        })
   572	      );
   573	    }
   574	    debounceTimer = null;
   575	  }, 150);
   576	}

   578	export function subscribeToDatabase(callback: (table: string) => void): () => void {
   579	  if (typeof window === 'undefined') return () => {};
   580	
   581	  const handler = (e: Event) => {
   582	    const custom = e as CustomEvent<{ table: string }>;
   583	    callback(custom.detail?.table || DB_TABLES.ALL);
   584	  };
   585	  window.addEventListener(DB_CHANGE_EVENT, handler);
   586	
   587	  let unsubscribed = false;
   588	  return () => {
   589	    if (unsubscribed) return;
   590	    unsubscribed = true;
   591	    window.removeEventListener(DB_CHANGE_EVENT, handler);
   592	  };
   593	}

   823	    const cleanOrgId = orgId.trim().toUpperCase();

   831	      ...(existing || { org_id: cleanOrgId, id: generateUUID(), name: 'Organization' }),
   832	      ...safeUpdates,
   833	    });
   834	
   835	    const orgsSnapshot = [...orgs];
   836	
   837	    // Optimistically save to cache & notify UI
   838	    if (idx !== -1) {
   839	      orgs[idx] = updated;
   840	    } else {
   841	      orgs.unshift(updated);
   842	    }

   858	            STORAGE_KEYS.ORGS,

   872	    } else if (isSupabaseConfigured()) {

   996	  // ------------------------------------------
   997	  async getEventsResult(orgId: string, force: boolean = false): Promise<ReadResult<SupabaseEvent[]>> {
   998	    const cleanOrgId = orgId.trim().toUpperCase();
   999	    const cacheKey = `${STORAGE_KEYS.EVENTS}_${cleanOrgId}`;
  1000	    const cachedEvents = this.getCachedEvents(cleanOrgId);
  1001	
  1002	    // 5-Minute SWR Cache Throttle: If cache is valid, return it immediately (even when empty)
  1003	    if (!force && isCacheValid(cacheKey)) {
  1004	      return { ok: true, data: cachedEvents, fromCache: true, error: null, timedOut: false };
  1005	    }
  1006	
  1007	    return coalesceRead(cacheKey, force, async () => {
  1008	      const latestCached = this.getCachedEvents(cleanOrgId);
  1009	
  1010	      if (isSupabaseConfigured()) {
  1011	        try {
  1012	          const res = await callSecuredReadRpc<any[]>('read_events', {});
  1013	          const data = Array.isArray(res.data) ? res.data : [];
  1014	          const normalized = data.map(normalizeEvent);
  1015	          const allCachedEvents = loadCache<SupabaseEvent>(STORAGE_KEYS.EVENTS, []);
  1016	          const otherOrgEvents = allCachedEvents.filter(
  1017	            (e) => (e.org_id || '').trim().toUpperCase() !== cleanOrgId
  1018	          );
  1019	          const changed = !areRecordsEqual(latestCached, normalized);
  1020	          saveCache(STORAGE_KEYS.EVENTS, [...normalized, ...otherOrgEvents], !changed);
  1021	          updateCacheTimestamp(cacheKey);
  1022	          return { ok: true, data: normalized, fromCache: false, error: null, timedOut: false };
  1023	        } catch (err: any) {
  1024	          const info = classifyReadError(err);
  1025	          console.warn('Supabase getEvents fallback:', info.message);
  1026	          return {
  1027	            ok: false,
  1028	            data: latestCached,
  1029	            fromCache: latestCached.length > 0,
  1030	            error: info.message,
  1031	            errorCode: info.code,
  1032	            timedOut: info.timedOut,
  1033	            unauthorized: info.unauthorized,
  1034	          };
  1035	        }
  1036	      } else {
  1037	        updateCacheTimestamp(cacheKey);
  1038	        return { ok: true, data: latestCached, fromCache: true, error: null, timedOut: false };
  1039	      }
  1040	    });
  1041	  },

  1191	  async updateEvent(eventId: string, updates: Partial<SupabaseEvent>): Promise<SupabaseEvent> {
  1192	    const events = loadCache<SupabaseEvent>(STORAGE_KEYS.EVENTS, []);
  1193	    const idx = events.findIndex((e) => e.id === eventId);
  1194	    const existing = idx !== -1 ? events[idx] : null;
  1195	
  1196	    // PRE-CHECK: If locked and the call touches a roster column without setting is_locked: false, throw typed error BEFORE optimistic write
  1197	    const touchesRoster =
  1198	      updates.assigned_participants !== undefined ||
  1199	      updates.assigned_judges !== undefined ||
  1200	      updates.participant_sl_numbers !== undefined;
  1201	
  1202	    if (existing?.is_locked && updates.is_locked !== false && touchesRoster) {
  1203	      const err = new Error(`ROSTER_LOCKED: Cannot modify roster on locked event "${existing.name}". Unlock event first.`);
  1204	      (err as any).code = 'ROSTER_LOCKED';
  1205	      throw err;
  1206	    }
  1207	
  1208	    const updated: SupabaseEvent = normalizeEvent({
  1209	      ...(existing || {}),
  1210	      ...updates,
  1211	      id: eventId,
  1212	      updated_at: new Date().toISOString(),
  1213	    });
  1214	
  1215	    // 1. OPTIMISTIC WRITE: Save to local cache immediately & notify UI
  1216	    if (idx !== -1) {
  1217	      events[idx] = updated;
  1218	    } else {
  1219	      events.unshift(updated);
  1220	    }
  1221	    saveCache(STORAGE_KEYS.EVENTS, events, false);
  1222	    if (updated.org_id) {
  1223	      invalidateCache(`${STORAGE_KEYS.EVENTS}_${updated.org_id.toUpperCase()}`);
  1224	    }
  1225	
  1226	    // 2. BACKGROUND CLOUD SYNC
  1227	    const dbUpdates: Record<string, any> = {
  1228	      ...updates,
  1229	      updated_at: updated.updated_at,
  1230	    };
  1231	    if (dbUpdates.rubric && !dbUpdates.rubric_schema) {
  1232	      dbUpdates.rubric_schema = dbUpdates.rubric;
  1233	    }
  1234	    delete dbUpdates.rubric;
  1235	
  1236	    if (isSupabaseConfigured() && isOnlineNavigator()) {
  1237	      try {
  1238	        const rpcRes = await callSecuredWriteRpc<SupabaseEvent>('event_update', {
  1239	          p_event_id: eventId,
  1240	          p_updates: dbUpdates,
  1241	        });
  1242	
  1243	        const syncedRow = rpcRes.data;
  1244	        if (syncedRow) {
  1245	          const synced = normalizeEvent({ ...syncedRow, is_archived: updated.is_archived });
  1246	          const currentEvents = loadCache<SupabaseEvent>(STORAGE_KEYS.EVENTS, []);
  1247	          saveCache(
  1248	            STORAGE_KEYS.EVENTS,
  1249	            currentEvents.map((e) => (e.id === synced.id ? synced : e)),
  1250	            true
  1251	          );
  1252	          notifyCloudSync('Event');
  1253	        }
  1254	      } catch (err: any) {
  1255	        const orgBlock = parseOrgEnforcementError(err);
  1256	        if (orgBlock) {
  1257	          eventBus.emit('org_enforcement', { code: orgBlock });
  1258	        }
  1259	        const isRosterLocked = Boolean(
  1260	          err?.code === 'ROSTER_LOCKED' ||
  1261	          err?.message?.includes('ROSTER_LOCKED') ||
  1262	          (err?.details && String(err.details).includes('ROSTER_LOCKED'))
  1263	        );
  1264	
  1265	        // If ROSTER_LOCKED, unauthorized, or any server-returned error: roll cache back and rethrow
  1266	        if (isServerRejectionError(err) || Boolean(err?.code) || isRosterLocked) {
  1267	          const currentEvents = loadCache<SupabaseEvent>(STORAGE_KEYS.EVENTS, []);
  1268	          if (existing) {
  1269	            const rollbackList = currentEvents.map((e) => (e.id === eventId ? existing : e));
  1270	            saveCache(STORAGE_KEYS.EVENTS, rollbackList, false);
  1271	          } else {
  1272	            const rollbackList = currentEvents.filter((e) => e.id !== eventId);
  1273	            saveCache(STORAGE_KEYS.EVENTS, rollbackList, false);
  1274	          }
  1275	          if (updated.org_id && err?.code !== 'unauthorized') {
  1276	            this.getEvents(updated.org_id, true).catch(() => {});
  1277	          }
  1278	          throw err;
  1279	        }
  1280	
  1281	        // Queue ONLY network errors
  1282	        console.warn('Offline mode: event update saved locally, deferring cloud sync:', err?.message || err);
  1283	        await queueMutationLocally('events', 'UPDATE', eventId, dbUpdates);
  1284	        notifyOfflineSaved('Event');
  1285	      }
  1286	    } else if (isSupabaseConfigured()) {
  1287	      await queueMutationLocally('events', 'UPDATE', eventId, dbUpdates);
  1288	      notifyOfflineSaved('Event');
  1289	    } else {
  1290	      notifyOfflineSaved('Event');
  1291	    }
  1292	
  1293	    return updated;
  1294	  },

  2051	  /**
  2052	   * Smart Deletion & Ledger Integrity Guardrail (Directive 3)
  2053	   * 1. Ended Events (Hard Block): If participant has finalized scores in ended/archived event, block deletion.

  2055	   * 3. No Events (Safe): Immediate clean deletion.
  2056	   */
  2057	  async checkParticipantDeletion(participantId: string): Promise<{
  2058	    tier: 'blocked' | 'warning' | 'safe';
  2059	    reason: string;
  2060	    eventNames: string[];
  2061	  }> {
  2062	    const allEvents = loadCache<SupabaseEvent>(STORAGE_KEYS.EVENTS, []);
  2063	    const allScores = this.getCachedScores();
  2064	
  2065	    // Tier 1: Ended Events Check (The Hard Block)
  2066	    const endedEventsWithParticipant = allEvents.filter((e) => {
  2067	      const isEnded = e.status === 'ended' || e.status === 'archived';
  2068	      if (!isEnded) return false;

  2070	      const hasScores = allScores.some((s) => s.participant_id === participantId && s.event_id === e.id);

  2079	      };
  2080	    }
  2081	
  2082	    // Tier 2: Ongoing/Draft Events Check (Warning & Override)
  2083	    const ongoingEvents = allEvents.filter((e) => {
  2084	      const isOngoing = e.status !== 'ended' && e.status !== 'archived';
  2085	      return isOngoing && (e.assigned_participants || []).includes(participantId);
  2086	    });
  2087	
  2088	    if (ongoingEvents.length > 0) {
  2089	      return {
  2090	        tier: 'warning',
  2091	        reason: 'ongoing_events',
  2092	        eventNames: ongoingEvents.map((e) => e.name),
  2093	      };
  2094	    }
  2095	
  2096	    // Tier 3: No Events (Safe)
  2097	    return {
  2098	      tier: 'safe',

  2383	  async submitScore(scoreData: {
  2384	    id?: string;
  2385	    event_id: string;
  2386	    judge_id: string;
  2387	    participant_id: string;
  2388	    score_data?: Record<string, number>;
  2389	    total_score: number;
  2390	    score_payload?: {
  2391	      criteria: Record<string, number>;
  2392	      total_score: number;
  2393	      submitted_at: string;
  2394	    };
  2395	  }): Promise<SupabaseScore> {
  2396	    // 1. Verify Event exists & is active
  2397	    const event = await this.getEventById(scoreData.event_id);
  2398	    if (!event) throw new Error('Event not found.');
  2399	    if (event.status !== 'active') {
  2400	      const statusLabel =
  2401	        event.status === 'ended' || event.status === 'archived'
  2402	          ? 'Event has ended.'
  2403	          : event.status === 'paused'
  2404	          ? 'Event is paused.'
  2405	          : `Event is ${event.status}.`;
  2406	      throw new Error(`${statusLabel} Ballot submission rejected.`);
  2407	    }
  2408	
  2409	    // 2. Verify Judge assigned
  2410	    const judge = await this.getJudgeById(scoreData.judge_id);
  2411	    if (!judge) throw new Error('Judge not found.');
  2412	
  2413	    const isAssigned =
  2414	      event.assigned_judges.includes(judge.id) ||
  2415	      (judge.assigned_events || []).includes(event.id) ||
  2416	      event.status === 'active';
  2417	    if (!isAssigned) {
  2418	      throw new Error('Judge is not assigned to evaluate this event.');
  2419	    }
  2420	
  2421	    // 3. Check if score already exists (Strict Immutability with 5-minute server-clock grace period support)
  2422	    const existing = await this.getScoreForJudgeParticipant(
  2423	      scoreData.event_id,
  2424	      scoreData.judge_id,
  2425	      scoreData.participant_id
  2426	    );
  2427	
  2428	    let isWithinGrace = false;
  2429	    if (existing) {
  2430	      const org = await this.getOrganizationByOrgId(event.org_id);
  2431	      const isGraceAllowed = Boolean(org?.grace_period_enabled);
  2432	      const diffMs = getServerNowMs() - new Date(existing.created_at).getTime();
  2433	      isWithinGrace = isGraceAllowed && diffMs <= 5 * 60 * 1000;
  2434	
  2435	      if (!isWithinGrace) {
  2436	        throw new Error(
  2437	          'RLS Permission Denied: Scores are immutable and cannot be updated once inserted.'
  2438	        );
  2439	      }
  2440	    }
  2441	
  2442	    const criteriaMap = scoreData.score_data || scoreData.score_payload?.criteria || {};
  2443	    const timestamp = new Date(getServerNowMs()).toISOString();
  2444	    const payloadJson = scoreData.score_payload || {
  2445	      criteria: criteriaMap,
  2446	      total_score: scoreData.total_score,
  2447	      submitted_at: timestamp,
  2448	    };
  2449	
  2450	    // 4. OPTIMISTIC WRITE: Keep snapshot so server rejections roll back cleanly
  2451	    const scoresSnapshot = loadCache<SupabaseScore>(STORAGE_KEYS.SCORES, []);
  2452	    let optimisticScore: SupabaseScore;
  2453	
  2454	    if (existing && isWithinGrace) {
  2455	      optimisticScore = normalizeScore({
  2456	        ...existing,
  2457	        score_data: criteriaMap,
  2458	        total_score: scoreData.total_score,
  2459	        score_payload: payloadJson,
  2460	      });
  2461	      const updated = scoresSnapshot.some((s) => s.id === existing.id)
  2462	        ? scoresSnapshot.map((s) => (s.id === existing.id ? optimisticScore : s))
  2463	        : [...scoresSnapshot, optimisticScore];
  2464	      saveCache(STORAGE_KEYS.SCORES, updated, false);
  2465	    } else {
  2466	      optimisticScore = normalizeScore({
  2467	        id: scoreData.id || generateUUID(),
  2468	        event_id: scoreData.event_id,
  2469	        judge_id: scoreData.judge_id,
  2470	        participant_id: scoreData.participant_id,
  2471	        score_data: criteriaMap,
  2472	        total_score: scoreData.total_score,
  2473	        score_payload: payloadJson,
  2474	        created_at: timestamp,
  2475	      });
  2476	      saveCache(STORAGE_KEYS.SCORES, [...scoresSnapshot, optimisticScore], false);
  2477	    }
  2478	
  2479	    invalidateCache(STORAGE_KEYS.SCORES);
  2480	
  2481	    // 5. CLOUD SYNC
  2482	    if (isSupabaseConfigured() && isOnlineNavigator()) {
  2483	      try {
  2484	        const scorePayload = {
  2485	          id: existing && isWithinGrace ? existing.id : optimisticScore.id,
  2486	          event_id: scoreData.event_id,
  2487	          judge_id: scoreData.judge_id,
  2488	          participant_id: scoreData.participant_id,
  2489	          total_score: scoreData.total_score,
  2490	          score_payload: payloadJson,
  2491	        };
  2492	
  2493	        const rpcRes = await callSecuredWriteRpc<SupabaseScore[]>('score_upsert_batch', {
  2494	          p_records: [scorePayload],
  2495	        });
  2496	
  2497	        const syncedRow = Array.isArray(rpcRes.data) ? rpcRes.data[0] : null;
  2498	        if (syncedRow) {
  2499	          const synced = normalizeScore(syncedRow);
  2500	          const currentScores = loadCache<SupabaseScore>(STORAGE_KEYS.SCORES, []);
  2501	          const existsInCache = currentScores.some((s) => s.id === synced.id);
  2502	          saveCache(
  2503	            STORAGE_KEYS.SCORES,
  2504	            existsInCache
  2505	              ? currentScores.map((s) => (s.id === synced.id ? synced : s))
  2506	              : [...currentScores, synced],
  2507	            true
  2508	          );
  2509	          notifyCloudSync('Score ballot');
  2510	          return synced;
  2511	        }
  2512	      } catch (err: any) {
  2513	        const orgBlock = parseOrgEnforcementError(err);
  2514	        if (orgBlock) {
  2515	          saveCache(STORAGE_KEYS.SCORES, scoresSnapshot, false);
  2516	          eventBus.emit('org_enforcement', { code: orgBlock });
  2517	          throw err;
  2518	        }
  2519	        if (isServerRejectionError(err)) {
  2520	          saveCache(STORAGE_KEYS.SCORES, scoresSnapshot, false);
  2521	          throw err;
  2522	        }
  2523	        console.warn('Offline mode: score ballot saved locally, deferring cloud sync:', err?.message || err);
  2524	        await queueScoreLocally(optimisticScore);
  2525	        notifyOfflineSaved('Score ballot');
  2526	      }
  2527	    } else if (isSupabaseConfigured()) {
  2528	      await queueScoreLocally(optimisticScore);
  2529	      notifyOfflineSaved('Score ballot');
  2530	    } else {
  2531	      notifyOfflineSaved('Score ballot');
  2532	    }
  2533	
  2534	    return optimisticScore;
  2535	  },

  2537	  async deleteScore(scoreId: string): Promise<boolean> {
  2538	    const scoresSnapshot = loadCache<SupabaseScore>(STORAGE_KEYS.SCORES, []);
  2539	    saveCache(
  2540	      STORAGE_KEYS.SCORES,
  2541	      scoresSnapshot.filter((s) => s.id !== scoreId),
  2542	      true
  2543	    );
  2544	    invalidateCache(STORAGE_KEYS.SCORES);
  2545	    if (isSupabaseConfigured() && isOnlineNavigator()) {
  2546	      try {
  2547	        await callSecuredWriteRpc('score_delete', { p_score_id: scoreId });
  2548	        notifyCloudSync('Score ballot removed');
  2549	        return true;

  2555	        console.warn('deleteScore cloud sync note:', err?.message || err);

  2557	    }
  2558	    return true;
