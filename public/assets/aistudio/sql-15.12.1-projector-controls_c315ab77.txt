    24	create or replace function public.org_update(
    25	  p_token text,
    26	  p_org_id text,
    27	  p_updates jsonb
    28	)
    29	returns jsonb
    30	language plpgsql
    31	security definer
    32	set search_path = public, extensions
    33	as $$
    34	declare
    35	  v_org_id text;
    36	  v_row public.organizations;
    37	  v_clean_ticker text;
    38	  v_interval int;

    40	  v_org_id := public._fj_verify_org_token(p_token, p_org_id);
    41	  if v_org_id is null then

    44	
    45	  if p_updates ? 'projector_ticker_text' then
    46	    v_clean_ticker := coalesce(p_updates->>'projector_ticker_text', '');

    48	    v_clean_ticker := replace(replace(v_clean_ticker, '<', ''), '>', '');
    49	    v_clean_ticker := regexp_replace(v_clean_ticker, '[\r\n\t]+', ' ', 'g');
    50	    v_clean_ticker := left(btrim(v_clean_ticker), 120);
    51	  end if;
    52	
    53	  if p_updates ? 'projector_carousel_interval' then
    54	    v_interval := greatest(15, least(180, coalesce((p_updates->>'projector_carousel_interval')::int, 60)));
    55	  end if;
    56	
    57	  update public.organizations
    58	  set
    59	    name = case when p_updates ? 'name' then coalesce(nullif(btrim(p_updates->>'name'), ''), name) else name end,
    60	    email = case when p_updates ? 'email' then nullif(btrim(p_updates->>'email'), '') else email end,
    61	    verification_status = case when p_updates ? 'verification_status' then coalesce(p_updates->>'verification_status', verification_status) else verification_status end,
    62	    tie_breaker_enabled = case when p_updates ? 'tie_breaker_enabled' then coalesce((p_updates->>'tie_breaker_enabled')::boolean, tie_breaker_enabled) else tie_breaker_enabled end,
    63	    tie_breaker_method = case when p_updates ? 'tie_breaker_method' then coalesce(p_updates->>'tie_breaker_method', tie_breaker_method) else tie_breaker_method end,
    64	    custom_participant_fields = case when p_updates ? 'custom_participant_fields' then coalesce(p_updates->'custom_participant_fields', custom_participant_fields) else custom_participant_fields end,
    65	    blind_judging_enabled = case when p_updates ? 'blind_judging_enabled' then coalesce((p_updates->>'blind_judging_enabled')::boolean, blind_judging_enabled) else blind_judging_enabled end,
    66	    auto_advance_enabled = case when p_updates ? 'auto_advance_enabled' then coalesce((p_updates->>'auto_advance_enabled')::boolean, auto_advance_enabled) else auto_advance_enabled end,
    67	    grace_period_enabled = case when p_updates ? 'grace_period_enabled' then coalesce((p_updates->>'grace_period_enabled')::boolean, grace_period_enabled) else grace_period_enabled end,
    68	    isolated_jury_enabled = case when p_updates ? 'isolated_jury_enabled' then coalesce((p_updates->>'isolated_jury_enabled')::boolean, isolated_jury_enabled) else isolated_jury_enabled end,
    69	    projector_blind_mode = case when p_updates ? 'projector_blind_mode' then coalesce((p_updates->>'projector_blind_mode')::boolean, projector_blind_mode) else projector_blind_mode end,
    70	    projector_spotlight_mode = case when p_updates ? 'projector_spotlight_mode' then coalesce((p_updates->>'projector_spotlight_mode')::boolean, projector_spotlight_mode) else projector_spotlight_mode end,
    71	    projector_ticker_text = case when p_updates ? 'projector_ticker_text' then coalesce(v_clean_ticker, '') else projector_ticker_text end,
    72	    projector_carousel_interval = case when p_updates ? 'projector_carousel_interval' then v_interval else projector_carousel_interval end,

    79	  end if;
    80	
    81	  return jsonb_build_object('ok', true, 'data', to_jsonb(v_row) - 'password');
    82	end;
    83	$$;

    88	-- 4. Update projector_fingerprint to reflect broadcast controls, label, event scope, and blank toggle
    89	create or replace function public.projector_fingerprint(p_view_token text)
    90	returns jsonb
    91	language plpgsql
    92	security definer
    93	set search_path = public, extensions
    94	as $$
    95	declare
    96	  v_pairing public.projector_pairings;
    97	  v_org_sig text;
    98	  v_evt_sig text;
    99	  v_part_sig text;
   100	  v_judge_sig text;
   101	  v_score_sig text;
   102	  v_version text;
   103	begin
   104	  v_pairing := public._fj_verify_projector_token(p_view_token);
   105	  if v_pairing.id is null then
   106	    return jsonb_build_object('ok', false, 'error', 'expired_or_revoked');
   107	  end if;
   108	
   109	  select md5(
   110	    coalesce(o.name, '') || '|' ||
   111	    coalesce(o.tie_breaker_enabled::text, 'false') || '|' ||
   112	    coalesce(o.tie_breaker_method, 'olympic') || '|' ||
   113	    coalesce(o.blind_judging_enabled::text, 'false') || '|' ||
   114	    coalesce(o.projector_blind_mode::text, 'false') || '|' ||
   115	    coalesce(o.projector_spotlight_mode::text, 'true') || '|' ||
   116	    coalesce(o.projector_ticker_text, '') || '|' ||
   117	    coalesce(o.projector_carousel_interval::text, '60') || '|' ||
   118	    coalesce(o.projector_confetti_at::text, '')
   119	  ) into v_org_sig
   120	  from public.organizations o
   121	  where upper(o.org_id) = upper(v_pairing.org_id)
   122	  limit 1;
   123	
   124	  select count(*)::text || ':' || coalesce(max(greatest(e.created_at, coalesce(e.updated_at, e.created_at)))::text, '0')
   125	  into v_evt_sig
   126	  from public.events e
   127	  where upper(e.org_id) = upper(v_pairing.org_id)
   128	    and (v_pairing.event_id is null or e.id::text = v_pairing.event_id);
   129	
   130	  select count(*)::text || ':' || coalesce(max(p.created_at)::text, '0')
   131	  into v_part_sig
   132	  from public.participants p
   133	  where upper(p.org_id) = upper(v_pairing.org_id);
   134	
   135	  select count(*)::text || ':' || coalesce(max(j.created_at)::text, '0')
   136	  into v_judge_sig
   137	  from public.judges j
   138	  where upper(j.org_id) = upper(v_pairing.org_id);
   139	
   140	  select count(*)::text || ':' || coalesce(max(s.created_at)::text, '0') || ':' || coalesce(sum(s.total_score)::text, '0')
   141	  into v_score_sig
   142	  from public.scores s
   143	  join public.events e on e.id = s.event_id
   144	  where upper(e.org_id) = upper(v_pairing.org_id)
   145	    and (v_pairing.event_id is null or e.id::text = v_pairing.event_id);
   146	
   147	  v_version := public._fj_projector_sha256(
   148	    coalesce(v_org_sig, '') || '|' ||
   149	    coalesce(v_evt_sig, '') || '|' ||
   150	    coalesce(v_part_sig, '') || '|' ||
   151	    coalesce(v_judge_sig, '') || '|' ||
   152	    coalesce(v_score_sig, '') || '|' ||
   153	    coalesce(v_pairing.event_id, '') || '|' ||
   154	    coalesce(v_pairing.label, '') || '|' ||
   155	    coalesce(v_pairing.is_blank::text, 'false')
   156	  );
   157	
   158	  return jsonb_build_object(
   159	    'ok', true,
   160	    'version', v_version,
   161	    'org_id', v_pairing.org_id,
   162	    'event_id', v_pairing.event_id,
   163	    'label', v_pairing.label,
   164	    'is_blank', coalesce(v_pairing.is_blank, false)
   165	  );
   166	end;
   167	$$;

   169	revoke all on function public.projector_fingerprint(text) from public;
   170	grant execute on function public.projector_fingerprint(text) to anon, authenticated;

   172	-- 5. Update projector_snapshot: enforce Score Secrecy (Blind Mode) on the server
   173	--    When projector_blind_mode is true:
   174	--      - scores contain ZERO point values (no total_score, no score_payload)
   175	--      - blind_ranks returns server-computed rank + judge_count only (no averages/totals)
   176	create or replace function public.projector_snapshot(p_view_token text)
   177	returns jsonb
   178	language plpgsql
   179	security definer
   180	set search_path = public, extensions
   181	as $$
   182	declare
   183	  v_pairing public.projector_pairings;
   184	  v_org_rec public.organizations;
   185	  v_org jsonb;
   186	  v_events jsonb;
   187	  v_judges jsonb;
   188	  v_participants jsonb;
   189	  v_scores jsonb;
   190	  v_blind_ranks jsonb := '[]'::jsonb;
   191	  v_scoped_event public.events;
   192	  v_is_blind boolean := false;
   193	  v_tie_enabled boolean := false;
   194	  v_tie_method text := 'olympic';
   195	begin
   196	  v_pairing := public._fj_verify_projector_token(p_view_token);
   197	  if v_pairing.id is null then
   198	    return jsonb_build_object('ok', false, 'error', 'expired_or_revoked');
   199	  end if;
   200	
   201	  select * into v_org_rec
   202	  from public.organizations o
   203	  where upper(o.org_id) = upper(v_pairing.org_id)
   204	  limit 1;
   205	
   206	  v_is_blind := coalesce(v_org_rec.projector_blind_mode, false);
   207	  v_tie_enabled := coalesce(v_org_rec.tie_breaker_enabled, false);
   208	  v_tie_method := coalesce(v_org_rec.tie_breaker_method, 'olympic');
   209	
   210	  v_org := jsonb_build_object(
   211	    'org_id', v_org_rec.org_id,
   212	    'name', v_org_rec.name,
   213	    'tie_breaker_enabled', v_tie_enabled,
   214	    'tie_breaker_method', v_tie_method,
   215	    'blind_judging_enabled', coalesce(v_org_rec.blind_judging_enabled, false),
   216	    'projector_blind_mode', v_is_blind,
   217	    'projector_spotlight_mode', coalesce(v_org_rec.projector_spotlight_mode, true),
   218	    'projector_ticker_text', left(coalesce(v_org_rec.projector_ticker_text, ''), 120),
   219	    'projector_carousel_interval', greatest(15, least(180, coalesce(v_org_rec.projector_carousel_interval, 60))),
   220	    'projector_confetti_at', v_org_rec.projector_confetti_at
   221	  );
   222	
   223	  if v_pairing.event_id is not null then
   224	    select * into v_scoped_event
   225	    from public.events e
   226	    where e.id::text = v_pairing.event_id
   227	      and upper(e.org_id) = upper(v_pairing.org_id)
   228	    limit 1;
   229	  end if;
   230	
   231	  select coalesce(
   232	    jsonb_agg(
   233	      jsonb_build_object(
   234	        'id', e.id,
   235	        'org_id', e.org_id,
   236	        'name', e.name,
   237	        'category', e.category,
   238	        'date', e.date,
   239	        'status', e.status,
   240	        'started_at', e.started_at,
   241	        'elapsed_seconds', e.elapsed_seconds,
   242	        'rubric_schema', coalesce(e.rubric_schema, '[]'::jsonb),
   243	        'assigned_judges', coalesce(e.assigned_judges, '{}'::text[]),
   244	        'assigned_participants', coalesce(e.assigned_participants, '{}'::text[]),
   245	        'participant_sl_numbers', coalesce(e.participant_sl_numbers, '{}'::jsonb),
   246	        'is_locked', coalesce(e.is_locked, false),
   247	        'is_archived', coalesce(e.is_archived, false),
   248	        'created_at', e.created_at,
   249	        'updated_at', e.updated_at
   250	      )
   251	      order by e.created_at desc
   252	    ),
   253	    '[]'::jsonb
   254	  ) into v_events
   255	  from public.events e
   256	  where upper(e.org_id) = upper(v_pairing.org_id)
   257	    and coalesce(e.is_archived, false) = false
   258	    and lower(coalesce(e.status, '')) <> 'archived'
   259	    and (v_pairing.event_id is null or e.id::text = v_pairing.event_id);
   260	
   261	  -- Safe Judge fields only (NO pin, NO current_session_token)
   262	  select coalesce(
   263	    jsonb_agg(
   264	      jsonb_build_object(
   265	        'id', j.id,
   266	        'org_id', j.org_id,
   267	        'name', j.name,
   268	        'judge_id', j.judge_id,
   269	        'assigned_events', coalesce(j.assigned_events, '{}'::text[]),
   270	        'created_at', j.created_at
   271	      )
   272	      order by j.created_at asc
   273	    ),
   274	    '[]'::jsonb
   275	  ) into v_judges
   276	  from public.judges j
   277	  where upper(j.org_id) = upper(v_pairing.org_id)
   278	    and (
   279	      v_pairing.event_id is null
   280	      or (
   281	        v_scoped_event.id is not null and (
   282	          j.id::text = any(coalesce(v_scoped_event.assigned_judges, '{}'::text[]))
   283	          or v_pairing.event_id = any(coalesce(j.assigned_events, '{}'::text[]))
   284	        )
   285	      )
   286	    );
   287	
   288	  -- Safe Participant fields only (NO mobile_number)
   289	  select coalesce(
   290	    jsonb_agg(
   291	      jsonb_build_object(
   292	        'id', p.id,
   293	        'org_id', p.org_id,
   294	        'identity_id', p.identity_id,
   295	        'sl_no', p.sl_no,
   296	        'name', p.name,
   297	        'school', p.school,
   298	        'class_or_age', p.class_or_age,
   299	        'custom_values', coalesce(p.custom_values, '{}'::jsonb),
   300	        'created_at', p.created_at
   301	      )
   302	      order by p.created_at asc
   303	    ),
   304	    '[]'::jsonb
   305	  ) into v_participants
   306	  from public.participants p
   307	  where upper(p.org_id) = upper(v_pairing.org_id)
   308	    and (
   309	      v_pairing.event_id is null
   310	      or (
   311	        v_scoped_event.id is not null and (
   312	          cardinality(coalesce(v_scoped_event.assigned_participants, '{}'::text[])) = 0
   313	          or p.id::text = any(v_scoped_event.assigned_participants)
   314	        )
   315	      )
   316	    );
   317	
   318	  if v_is_blind then
   319	    -- Server-enforced Score Secrecy: strip total_score & score_payload completely
   320	    select coalesce(
   321	      jsonb_agg(
   322	        jsonb_build_object(
   323	          'id', s.id,
   324	          'event_id', s.event_id,
   325	          'judge_id', s.judge_id,
   326	          'participant_id', s.participant_id,
   327	          'created_at', s.created_at
   328	        )
   329	        order by s.created_at asc
   330	      ),
   331	      '[]'::jsonb
   332	    ) into v_scores
   333	    from public.scores s
   334	    join public.events e on e.id = s.event_id
   335	    where upper(e.org_id) = upper(v_pairing.org_id)
   336	      and coalesce(e.is_archived, false) = false
   337	      and lower(coalesce(e.status, '')) <> 'archived'
   338	      and (v_pairing.event_id is null or e.id::text = v_pairing.event_id);
   339	
   340	    -- Server-computed relative ranks without any numeric point values
   341	    with per_participant as (
   342	      select
   343	        s.event_id,
   344	        s.participant_id,
   345	        count(*)::int as judge_count,
   346	        case
   347	          when v_tie_enabled and v_tie_method = 'olympic' and count(*) >= 3 then
   348	            round(((sum(s.total_score) - max(s.total_score) - min(s.total_score))::numeric / (count(*) - 2)), 2)
   349	          when v_tie_enabled and v_tie_method = 'manual' then
   350	            round(avg(s.total_score)::numeric, 4)
   351	          else
   352	            round(avg(s.total_score)::numeric, 2)
   353	        end as avg_val
   354	      from public.scores s
   355	      join public.events e on e.id = s.event_id
   356	      where upper(e.org_id) = upper(v_pairing.org_id)
   357	        and coalesce(e.is_archived, false) = false
   358	        and lower(coalesce(e.status, '')) <> 'archived'
   359	        and (v_pairing.event_id is null or e.id::text = v_pairing.event_id)
   360	      group by s.event_id, s.participant_id
   361	    ),
   362	    ranked as (
   363	      select
   364	        pp.event_id,
   365	        pp.participant_id,
   366	        pp.judge_count,
   367	        case
   368	          when not v_tie_enabled or v_tie_method = 'shared' then
   369	            rank() over (partition by pp.event_id order by pp.avg_val desc, pp.participant_id asc)
   370	          else
   371	            row_number() over (partition by pp.event_id order by pp.avg_val desc, pp.participant_id asc)
   372	        end as computed_rank
   373	      from per_participant pp
   374	    )
   375	    select coalesce(
   376	      jsonb_agg(
   377	        jsonb_build_object(
   378	          'event_id', r.event_id,
   379	          'participant_id', r.participant_id,
   380	          'rank', r.computed_rank,
   381	          'judge_count', r.judge_count
   382	        )
   383	        order by r.event_id, r.computed_rank asc
   384	      ),
   385	      '[]'::jsonb
   386	    ) into v_blind_ranks
   387	    from ranked r;
   388	  else
   389	    select coalesce(
   390	      jsonb_agg(
   391	        jsonb_build_object(
   392	          'id', s.id,
   393	          'event_id', s.event_id,
   394	          'judge_id', s.judge_id,
   395	          'participant_id', s.participant_id,
   396	          'score_payload', s.score_payload,
   397	          'total_score', s.total_score,
   398	          'created_at', s.created_at
   399	        )
   400	        order by s.created_at asc
   401	      ),
   402	      '[]'::jsonb
   403	    ) into v_scores
   404	    from public.scores s
   405	    join public.events e on e.id = s.event_id
   406	    where upper(e.org_id) = upper(v_pairing.org_id)
   407	      and coalesce(e.is_archived, false) = false
   408	      and lower(coalesce(e.status, '')) <> 'archived'
   409	      and (v_pairing.event_id is null or e.id::text = v_pairing.event_id);
   410	  end if;
   411	
   412	  return jsonb_build_object(
   413	    'ok', true,
   414	    'org', v_org,
   415	    'events', v_events,
   416	    'judges', v_judges,
   417	    'participants', v_participants,
   418	    'scores', v_scores,
   419	    'blind_ranks', v_blind_ranks,
   420	    'scoped_event_id', v_pairing.event_id,
   421	    'label', v_pairing.label,
   422	    'is_blank', coalesce(v_pairing.is_blank, false)
   423	  );
   424	end;
   425	$$;

   430	-- 6. Update projector_list to include is_blank, 60-second online flag, and live_count
   431	create or replace function public.projector_list(
   432	  p_token text,
   433	  p_org_id text
   434	)
   435	returns jsonb
   436	language plpgsql
   437	security definer
   438	set search_path = public, extensions
   439	as $$
   440	declare
   441	  v_org_id text;
   442	  v_list jsonb;
   443	  v_live_count int := 0;
   444	begin
   445	  v_org_id := public._fj_verify_org_token(p_token, p_org_id);
   446	  if v_org_id is null then
   447	    return jsonb_build_object(
   448	      'ok', false,
   449	      'error', 'unauthorized',
   450	      'message', 'Session expired or invalid. Please sign in again.'
   451	    );
   452	  end if;
   453	
   454	  update public.projector_pairings
   455	  set status = 'expired'
   456	  where upper(org_id) = upper(v_org_id)
   457	    and status = 'approved'
   458	    and coalesce(last_seen_at, approved_at, created_at) < (now() - interval '24 hours');
   459	
   460	  select count(*)::int into v_live_count
   461	  from public.projector_pairings p
   462	  where upper(p.org_id) = upper(v_org_id)
   463	    and p.status = 'approved'
   464	    and p.last_seen_at is not null
   465	    and p.last_seen_at >= (now() - interval '60 seconds');
   466	
   467	  select coalesce(
   468	    jsonb_agg(
   469	      jsonb_build_object(
   470	        'id', p.id,
   471	        'label', coalesce(p.label, 'Projector Screen'),
   472	        'event_id', p.event_id,
   473	        'event_name', e.name,
   474	        'is_blank', coalesce(p.is_blank, false),
   475	        'is_online', (p.last_seen_at is not null and p.last_seen_at >= (now() - interval '60 seconds')),
   476	        'created_at', p.created_at,
   477	        'approved_at', p.approved_at,
   478	        'last_seen_at', p.last_seen_at
   479	      )
   480	      order by coalesce(p.last_seen_at, p.approved_at, p.created_at) desc
   481	    ),
   482	    '[]'::jsonb
   483	  ) into v_list
   484	  from public.projector_pairings p
   485	  left join public.events e on e.id::text = p.event_id and upper(e.org_id) = upper(v_org_id)
   486	  where upper(p.org_id) = upper(v_org_id)
   487	    and p.status = 'approved';
   488	
   489	  return jsonb_build_object('ok', true, 'pairings', v_list, 'live_count', v_live_count);
   490	end;
   491	$$;

   496	-- 7. New RPC: projector_update (rename screen, change event scope, toggle blank screen)
   497	create or replace function public.projector_update(
   498	  p_token text,
   499	  p_org_id text,
   500	  p_pairing_id uuid,
   501	  p_label text default null,
   502	  p_event_id text default null,
   503	  p_set_event boolean default false,
   504	  p_is_blank boolean default null
   505	)
   506	returns jsonb
   507	language plpgsql
   508	security definer
   509	set search_path = public, extensions
   510	as $$
   511	declare
   512	  v_org_id text;
   513	  v_clean_event_id text;
   514	  v_clean_label text;
   515	  v_row public.projector_pairings;
   516	begin
   517	  v_org_id := public._fj_verify_org_token(p_token, p_org_id);
   518	  if v_org_id is null then
   519	    return jsonb_build_object(
   520	      'ok', false,
   521	      'error', 'unauthorized',
   522	      'message', 'Session expired or invalid. Please sign in again.'
   523	    );
   524	  end if;
   525	
   526	  if p_set_event then
   527	    v_clean_event_id := nullif(btrim(coalesce(p_event_id, '')), '');
   528	    if v_clean_event_id is not null then
   529	      if not exists (
   530	        select 1 from public.events e
   531	        where e.id::text = v_clean_event_id
   532	          and upper(e.org_id) = upper(v_org_id)
   533	      ) then
   534	        return jsonb_build_object(
   535	          'ok', false,
   536	          'error', 'invalid_event',
   537	          'message', 'Selected event was not found for this organization.'
   538	        );
   539	      end if;
   540	    end if;
   541	  end if;
   542	
   543	  if p_label is not null then
   544	    v_clean_label := regexp_replace(coalesce(p_label, ''), '<[^>]*>', '', 'g');
   545	    v_clean_label := left(coalesce(nullif(btrim(v_clean_label), ''), 'Projector Screen'), 60);
   546	  end if;
   547	
   548	  update public.projector_pairings
   549	  set
   550	    label = case when p_label is not null then v_clean_label else label end,
   551	    event_id = case when p_set_event then v_clean_event_id else event_id end,
   552	    is_blank = case when p_is_blank is not null then p_is_blank else is_blank end
   553	  where id = p_pairing_id
   554	    and upper(org_id) = upper(v_org_id)
   555	    and status = 'approved'
   556	  returning * into v_row;
   557	
   558	  if v_row.id is null then
   559	    return jsonb_build_object(
   560	      'ok', false,
   561	      'error', 'not_found',
   562	      'message', 'Paired screen not found.'
   563	    );
   564	  end if;
   565	
   566	  return jsonb_build_object(
   567	    'ok', true,
   568	    'pairing_id', v_row.id,
   569	    'label', coalesce(v_row.label, 'Projector Screen'),
   570	    'event_id', v_row.event_id,
   571	    'is_blank', coalesce(v_row.is_blank, false)
   572	  );
   573	end;
   574	$$;
   575	
   576	revoke all on function public.projector_update(text, text, uuid, text, text, boolean, boolean) from public;
   577	grant execute on function public.projector_update(text, text, uuid, text, text, boolean, boolean) to anon, authenticated;
